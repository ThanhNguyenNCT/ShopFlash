      ($(current page.north west)+(\pgfkeysvalueof{/pageframe/outer left},-\pgfkeysvalueof{/pageframe/outer top})$)

      ($(current page.south east)+(-\pgfkeysvalueof{/pageframe/outer right},\pgfkeysvalueof{/pageframe/outer bottom})$);
